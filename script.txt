        videoEl.muted = true;
        videoEl.play();

        myMediaRecorder = new MediaRecorder(stream);
        audioChunks = [];

        myMediaRecorder.ondataavailable = e => audioChunks.push(e.data);
        myMediaRecorder.onstop = () => {
          const blob = new Blob(audioChunks, { type: 'audio/webm' });
          const reader = new FileReader();
          reader.readAsDataURL(blob);
          reader.onloadend = () => {
            socket.emit('stop_recording_sync', { audioBase64: reader.result, duration: (turn.endTime - turn.startTime) });
          };
        };

        myMediaRecorder.start();
        isRecording = true;
        document.getElementById('rec-text').innerText = 'جاري تسجيل صوت شخصيتك... 🔴';
        document.getElementById('rec-btn').classList.replace('bg-red-600', 'bg-amber-600');

        videoEl.muted = true;
        videoEl.play();
        document.getElementById('rec-text').innerText = `جاري دوبلاج شخصية (${turn.characterName})... 🔴`;
      }

      const durationMs = (turn.endTime - turn.startTime) * 1000;
      clearTimeout(turnTimer);
      turnTimer = setTimeout(() => {
        videoEl.pause();
        if (isRecording && recorderSocketId === mySocketId) {
          myMediaRecorder.stop();
          isRecording = false;
        }
      }, durationMs);

    // ===== التحليل التلقائي للمشاهد الطويلة + تثبيت هوية المتحدث =====
    // ملاحظة: هذا ليس Speaker-ID سحرياً؛ هو بصمة صوت محلية تعتمد على خصائص الطيف
    // وتستخدم تجميعاً متماسكاً بدلاً من (المقطع 1 = شخصية 1، المقطع 2 = شخصية 2...).
    async function runAutoSceneDetection() {
      if (!uploadedCustomVideoUrl || !videoEl.src) return alert('ارفع الفيديو أولاً.');
      const status = document.getElementById('auto-status');
      status.textContent = '⏳ جاري اكتشاف الكلام ثم استخراج بصمات الصوت...';
      try {
        const stream = videoEl.captureStream ? videoEl.captureStream() : videoEl.mozCaptureStream?.();
        if (!stream || !stream.getAudioTracks().length) throw new Error('المتصفح لا يدعم قراءة مسار صوت الفيديو.');

        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const source = audioCtx.createMediaStreamSource(stream);
        const analyser = audioCtx.createAnalyser();
        analyser.fftSize = 2048;
        analyser.smoothingTimeConstant = 0.2;
        source.connect(analyser);
        const data = new Uint8Array(analyser.fftSize);
        const freq = new Uint8Array(analyser.frequencyBinCount);

        const wasMuted = videoEl.muted;
        const oldTime = videoEl.currentTime;
        videoEl.muted = true;
        videoEl.currentTime = 0;
        await videoEl.play();

        // 1) اكتشاف الكلام عبر RMS مع threshold تكيفي.
        const samples = [];
        const started = performance.now();
        let raf;
        await new Promise(resolve => {
          const tick = () => {
            analyser.getByteTimeDomainData(data);
            let sum = 0;
            for (let i=0;i<data.length;i++) { const v=(data[i]-128)/128; sum += v*v; }
            samples.push({ t: videoEl.currentTime, rms: Math.sqrt(sum/data.length) });
            if (videoEl.ended || performance.now()-started > Math.max(30000,(videoEl.duration||300)*1100)) return resolve();
            raf=requestAnimationFrame(tick);
          };
          tick();
        });
        cancelAnimationFrame(raf);
        videoEl.pause();
        videoEl.currentTime = oldTime;
        videoEl.muted = wasMuted;

        const rmsVals=samples.map(x=>x.rms).sort((a,b)=>a-b);
        const noise=rmsVals[Math.floor(rmsVals.length*0.2)] || 0.01;
        const threshold=Math.max(0.012, noise*1.65);
        const raw=samples.filter(x=>x.rms>threshold);
        const segments=[];
        if(raw.length){
          let start=raw[0].t, prev=raw[0].t;
          for(const x of raw.slice(1)){
            if(x.t-prev>0.38){
              if(prev-start>=0.35) segments.push({startTime:Math.max(0,start-0.10),endTime:Math.min(videoEl.duration,prev+0.16)});
              start=x.t;
            }
            prev=x.t;
          }
          if(prev-start>=0.35) segments.push({startTime:Math.max(0,start-0.10),endTime:Math.min(videoEl.duration,prev+0.16)});
        }

        const merged=[];
        for(const seg of segments){
          const lastSeg=merged[merged.length-1];
          if(lastSeg && seg.startTime-lastSeg.endTime<0.30) lastSeg.endTime=seg.endTime;
          else merged.push({...seg});
        }

        if (!merged.length) throw new Error('لم يتم العثور على كلام واضح. جرّب مقطعاً بصوت أوضح.');

        // 2) بصمة صوت لكل مقطع. نأخذ عدة لقطات من الجزء الأوسط، وليس لقطة واحدة.
        const fingerprints=[];
        const wait = ms => new Promise(r=>setTimeout(r,ms));
        const makeFingerprint = () => {
          analyser.getByteTimeDomainData(data);
          analyser.getByteFrequencyData(freq);
          let energy=0, zc=0;
          for(let i=0;i<data.length;i++){
            const v=(data[i]-128)/128; energy += v*v;
            if(i>0 && ((data[i]-128)*(data[i-1]-128)<0)) zc++;
          }
          let weighted=0, total=0, high=0, low=0;
          for(let i=0;i<freq.length;i++){
            const v=freq[i]/255, f=i*audioCtx.sampleRate/(analyser.fftSize*2);
            weighted += f*v; total += v;
            if(f<1000) low += v;
            if(f>2500) high += v;
          }
          const centroid = total ? weighted/total : 0;
          return [
            Math.log10(1e-5 + energy/data.length),
            zc/data.length,
            centroid/4000,
            low/(total+1e-6),
            high/(total+1e-6)
          ];
        };

        for(let i=0;i<merged.length && i<300;i++){
          const seg=merged[i];
          videoEl.currentTime = Math.min(videoEl.duration-0.05, seg.startTime + Math.max(0.08,(seg.endTime-seg.startTime)*0.35));
          await videoEl.play();
          const frames=[];
          const sampleUntil=performance.now()+Math.min(650,Math.max(250,(seg.endTime-seg.startTime)*500));
          while(performance.now()<sampleUntil){
            frames.push(makeFingerprint());
            await wait(35);
          }
          videoEl.pause();
          const avg=frames[0].map((_,j)=>frames.reduce((a,f)=>a+f[j],0)/frames.length);
          fingerprints.push({ ...seg, feature:avg });
          status.textContent=`🎙️ بصمة الصوت: ${i+1}/${Math.min(merged.length,300)}`;
        }

        // 3) توحيد المقياس ثم clustering. عدد المتحدثين يختار من البيانات نفسها،
        // مع عقوبة تمنع تقسيم صوت واحد إلى شخصيات كثيرة بسبب اختلاف النبرة.
        const means = fingerprints[0].feature.map((_,j)=>fingerprints.reduce((a,x)=>a+x.feature[j],0)/fingerprints.length);
        const stds = means.map((m,j)=>Math.sqrt(fingerprints.reduce((a,x)=>a+Math.pow(x.feature[j]-m,2),0)/fingerprints.length)+1e-6);
        const vecs=fingerprints.map(x=>x.feature.map((v,j)=>(v-means[j])/stds[j]));
        const dist=(a,b)=>Math.sqrt(a.reduce((sum,v,j)=>sum+Math.pow(v-b[j],2),0));

        // Agglomerative clustering: لا ننشئ مجموعة جديدة إلا إذا كان الصوت مختلفاً بوضوح.
        const clusters=[];
        const nearest=[];
        vecs.forEach((v,i)=>{
          if(!clusters.length){ clusters.push({center:[...v], members:[i]}); return; }
          let best=-1, bestD=Infinity;
          clusters.forEach((c,ci)=>{ const d=dist(v,c.center); if(d<bestD){bestD=d;best=ci;} });
          nearest.push(bestD);
          // threshold محافظ. إذا تغيرت النبرة قليلاً يبقى نفس المتحدث.
          const adaptiveThreshold = 1.05;
          if(bestD <= adaptiveThreshold || clusters.length>=6){
            const c=clusters[best]; c.members.push(i);
            c.center=c.center.map((x,j)=>x+(v[j]-x)/c.members.length);
          } else clusters.push({center:[...v],members:[i]});
        });

        // دمج المجموعات الصغيرة إذا كانت أقرب لمجموعة أخرى، لمنع ظهور شخصية ثالثة بسبب مقطع شاذ.
        for(let pass=0;pass<3;pass++){
          for(let ci=clusters.length-1;ci>=0;ci--){
            const c=clusters[ci];
            if(c.members.length>1) continue;
            let best=-1,bestD=Infinity;
            clusters.forEach((other,oi)=>{ if(oi===ci)return; const d=dist(c.center,other.center); if(d<bestD){bestD=d;best=oi;} });
            if(best>=0 && bestD<1.45){
              const target=clusters[best];
              c.members.forEach(idx=>target.members.push(idx));
              target.center=target.members.reduce((acc,idx)=>acc.map((x,j)=>x+vecs[idx][j]),[0,0,0,0,0]).map(x=>x/target.members.length);
              clusters.splice(ci,1);
            }
          }
        }

        // إذا دمج التجميع كل شيء في صوت واحد، نختبر وجود فصل واضح بين صوتين قبل الاستسلام.
        // هذا يمنع حالة: "شخصية واحدة فقط" في أغلب الحوارات، مع الحفاظ على صوت واحد إذا لم توجد إشارة كافية.
        if (clusters.length === 1 && vecs.length >= 4) {
          let a=0,b=1,maxD=0;
          for(let i=0;i<vecs.length;i++) for(let j=i+1;j<vecs.length;j++){ const d=dist(vecs[i],vecs[j]); if(d>maxD){maxD=d;a=i;b=j;} }
          if(maxD > 1.55){
            let c1=[...vecs[a]], c2=[...vecs[b]], g1=[], g2=[];
            for(let pass=0;pass<6;pass++){
              g1=[];g2=[];
              vecs.forEach((v,i)=>(dist(v,c1)<=dist(v,c2)?g1:g2).push(i));
              if(!g1.length||!g2.length) break;
              c1=c1.map((_,j)=>g1.reduce((sum,i)=>sum+vecs[i][j],0)/g1.length);
              c2=c2.map((_,j)=>g2.reduce((sum,i)=>sum+vecs[i][j],0)/g2.length);
            }
            if(g1.length && g2.length){ clusters.length=0; clusters.push({center:c1,members:g1},{center:c2,members:g2}); }
          }
        }

        const labels=new Array(fingerprints.length);
        clusters.forEach((c,ci)=>c.members.forEach(idx=>labels[idx]=ci+1));
        autoDetectedSegments=fingerprints.map((s,i)=>({
          index:i,startTime:+s.startTime.toFixed(2),endTime:+s.endTime.toFixed(2),
          characterName:`الشخصية ${labels[i]}`,
          speakerCluster:labels[i]
        }));
        customSegments=autoDetectedSegments;
        renderCustomSegmentsList();
        const chars=[...new Set(customSegments.map(s=>s.characterName))];
        status.textContent=`✅ تم اكتشاف ${customSegments.length} حوار وربطها بـ ${chars.length} أصوات ثابتة. إذا كان صوت الشخصية يعود لاحقاً سيبقى على نفس الشخصية.`;
        alert(`تم التحليل. تم اكتشاف ${chars.length} شخصيات صوتية تقريبية، وليس شخصية جديدة لكل رجعة للكلام. راجع النتائج قبل النشر.`);
      } catch(err) {
        console.error(err);
        document.getElementById('auto-status').textContent='⚠️ تعذر تحليل بصمات المتحدثين في هذا المتصفح.';
        alert('تعذر التحليل التلقائي: '+err.message);
      } finally {
        try { source.disconnect(); analyser.disconnect(); await audioCtx.close(); } catch(e) {}
      }
    }

    function showDubResultScore(recordings, turns) {
      const panel=document.getElementById('result-panel');
      panel.classList.remove('hidden');
      const entries=turns.map((t,i)=>({t,rec:recordings[i]})).filter(x=>x.rec);
      if(!entries.length){ document.getElementById('match-score').textContent='0%'; return; }
      let total=0;
      entries.forEach(({t,rec})=>{
        const expected=Math.max(0.2,t.endTime-t.startTime);
        const recLen=rec.duration || expected;
        const durationRatio=Math.min(expected/recLen,recLen/expected);
        total += Math.max(0,Math.min(1,durationRatio))*0.65 + 0.35;
      });
      const coverage=entries.length/turns.length;
      const score=Math.round(Math.max(0,Math.min(100,(total/entries.length)*coverage*100)));
      document.getElementById('match-score').textContent=score+'%';
      document.getElementById('score-details').textContent=`تغطية الحوارات: ${Math.round(coverage*100)}% • الدرجة تجمع تغطية الأدوار وتقارب مدة التسجيل من مدة الحوار.`;
      document.getElementById('render-result-btn').classList.remove('hidden');
    }

    function formatTime(sec) {
      const m = Math.floor(sec / 60);
      const s = Math.floor(sec % 60);
      const ms = Math.floor((sec % 1) * 10);
      return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}.${ms}`;
